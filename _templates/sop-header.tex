% =============================================================================
% SOP PDF Header/Footer Configuration
% =============================================================================
% This template reads the following YAML metadata from each SOP:
%   sop-number:  e.g., "SOP-L29"
%   revision:    e.g., "2.0"
% They are injected by Quarto/Pandoc as \$sop-number\$ and \$revision\$.
% We store them in LaTeX macros for use in headers and footers.
% =============================================================================

\usepackage{fancyhdr}
\usepackage{lastpage}
\usepackage{xcolor}

% Define EA Program colors
\definecolor{eablue}{HTML}{0057B8}

% ---------------------------------------------------------------------------
% Capture YAML metadata as LaTeX macros (with fallback defaults)
% ---------------------------------------------------------------------------
$if(sop-number)$
  \newcommand{\sopnumber}{$sop-number$}
$else$
  \newcommand{\sopnumber}{SOP-XXX}
$endif$

$if(revision)$
  \newcommand{\soprevision}{$revision$}
$else$
  \newcommand{\soprevision}{--}
$endif$

% ---------------------------------------------------------------------------
% Header and Footer
% ---------------------------------------------------------------------------
\pagestyle{fancy}
\fancyhf{}

% Header
\fancyhead[L]{%
  \small\textcolor{eablue}{\textbf{EA Program -- Pomona College}}%
}
\fancyhead[R]{%
  \small\textcolor{eablue}{\sopnumber}%
}

% Footer
\fancyfoot[L]{\small\textit{Rev.\ \soprevision\ (\today)}}
\fancyfoot[C]{\small\textsc{For Internal Use Only}}
\fancyfoot[R]{\small Page \thepage\ of \pageref{LastPage}}

% Header/footer rules
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{eablue}\leaders\hrule height \headrulewidth\hfill}}
\renewcommand{\footrule}{\hbox to\headwidth{\color{eablue}\leaders\hrule height \footrulewidth\hfill}}

% ---------------------------------------------------------------------------
% First page style (title page)
% ---------------------------------------------------------------------------
\fancypagestyle{plain}{%
  \fancyhf{}
  \fancyhead[L]{%
    \textcolor{eablue}{\textbf{EA Program -- Pomona College}}%
  }
  \fancyhead[R]{\small\textcolor{eablue}{\sopnumber}}
  \fancyfoot[L]{\small\textit{Rev.\ \soprevision\ (\today)}}
  \fancyfoot[C]{\small\textsc{For Internal Use Only}}
  \fancyfoot[R]{\small Page \thepage\ of \pageref{LastPage}}
  \renewcommand{\headrulewidth}{0.4pt}
  \renewcommand{\footrulewidth}{0.4pt}
}
